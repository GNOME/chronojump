1. Install Homebrew

/bin/bash -c "$(curl -fsSL https://raw.githubcontent.com/Homebrew/install/HEAD/install.sh)"
brew update

2. Execute the following commands (Only when the first time. Please replace "~/Desktop" with the real path of chronojump.)

brew install git
brew install automake
brew install autoconf
brew install libtool
brew install intltool
brew install pkg-config
brew install gettext
brew install glib
brew install python3
brew install gtk+3

brew install mono
cd "~/Desktop/chronojump"
git clone -b main https://github.com/mono/gtk-sharp.git
//Open gtk-sharp/gtk/gui-thread-check/profiler/gui-thread-check.c and comment LINE 98-100
cd gtk-sharp
sh autogen.sh --prefix=`pkg-config --variable=prefix mono`
make
make install

cd ../package/macos
sh gtk-osx-setup.sh
export PATH="$PATH:~/.new_local/bin"
jhbuild bootstrap-gtk-osx
jhbuild build meta-gtk-osx-bootstrap meta-gtk-osx-gtk3

3. Execute the following commands (Only when the first time, when autotools files changed or when git tag created. Please replace "~/Desktop" with the real path of chronojump.)

cd "~/Desktop/chronojump"
sh autogen.sh --prefix ~/Desktop/chronojump/package/macos/app/Chronojump.app/Contents
make
make install

4. When no git tags, please modify "src/AssemblyInfo.cs" to add version number like 1.0.0.0:

[assembly: AssemblyVersion("1.0.0.0")]

5. Make a package (Only when a package required)

cd "~/Desktop/chronojump/package/macos"
make
